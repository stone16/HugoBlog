{{- /* Return the destination URL for a page published on stometa.top. This is intentionally a soft redirect: GitHub Pages cannot emit a server-side 301 for arbitrary paths, so the old URL remains the document that search engines crawl while the canonical points at stometa.dev. */ -}} {{- $path := .RelPermalink -}} {{- $destinationPath := $path -}} {{- if eq $path "/" -}} {{- $destinationPath = "/" -}} {{- else if eq $path "/posts/" -}} {{- $destinationPath = "/posts/" -}} {{- else if hasPrefix $path "/posts/" -}} {{- /* NOTE: Hugo reverses Go's argument order -- the prefix/suffix comes FIRST, then the string. Writing it the Go way silently yields "/" and every English post then falls through to a zh URL that 404s. */ -}} {{- $slug := strings.TrimSuffix "/" (strings.TrimPrefix "/posts/" $path) -}} {{- if or (eq $slug "confirmation-bias") (eq $slug "flywheel-effect") (eq $slug "google-ucp-goes-live-the-search-to-checkout-era-begins") -}} {{- $destinationPath = printf "/en/posts/%s/" $slug -}} {{- end -}} {{- end -}} {{- if ne $destinationPath "/" -}} {{- $destinationPath = strings.TrimSuffix "/" $destinationPath -}} {{- end -}} {{- return (printf "https://stometa.dev%s" $destinationPath) -}}
